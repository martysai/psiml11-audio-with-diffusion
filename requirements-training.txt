# Extra dependencies for src/audioseal_robust (generator-only robustness
# fine-tuning + evaluation), on top of requirements.txt. Not needed to use
# the core audioseal inference package.
#
# This file covers the identity and sgmse attacks. The two held-out
# diffusion attacks each need their own extra file on top of this one:
#   requirements-audioldm.txt    -- audioldm (AudioLDM latent diffusion)
#   requirements-mbd.txt         -- mbd (MultiBand Diffusion)

torchaudio>=2.1.0
soundfile               # data.py's WavDirDataset: avoids torchaudio.load's torchcodec/FFmpeg
                        # requirement (recent torchaudio routes .load() through torchcodec
                        # unconditionally, which needs a system FFmpeg install)
mlflow>=2.0.0          # default experiment tracking backend (local, no account needed)
wandb                  # optional alternative backend: set tracking.backend=wandb
torchmetrics            # evaluate.py: SI-SNR, PESQ (see metrics.py)
pesq                    # backing library for torchmetrics' PESQ wrapper
matplotlib              # evaluate.py: confusion-matrix + robustness-curve plots (see plotting.py)
tqdm                    # evaluate.py: per-attack progress bars (diffusion attacks run minutes per batch)

# BigVGAN / DAC attack backbones are NOT listed here -- they are stubbed
# with NotImplementedError + setup instructions in src/audioseal_robust/
# attacks.py until checkpoints/package versions are picked.

# SGMSEAttack needs src/sgmse/ (vendored, see that directory's VENDORED.md).
# pytorch_lightning/torch-ema/torchsde/torch-pesq/torchinfo/pystoi are its own
# inference-time deps beyond what's already listed above (librosa and scipy
# are also imported by src/sgmse/model.py, but already pulled in transitively
# elsewhere in this project); gdown is only needed to fetch a checkpoint from
# their Google Drive links, not at runtime.
gdown
pytorch_lightning
torch-ema
torchsde
torch-pesq
torchinfo
pystoi
librosa               # sgmse/util/other.py, data_module.py
scipy                 # sgmse/sampling, sdes.py
pytorch_lightning     # sgmse/model.py: ScoreModel is a LightningModule
